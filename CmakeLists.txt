cmake_minimum_required(VERSION 3.16)
project(dashtype LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

include(FetchContent)
FetchContent_Declare(
    ftxui
    GIT_REPOSITORY https://github.com/ArthurSonzogni/FTXUI.git
    GIT_TAG        v5.0.0
)
FetchContent_MakeAvailable(ftxui)

add_executable(dashtype src/main.cpp src/cli_parser.cpp src/text_buffer.cpp src/typing_engine.cpp src/metrics.cpp src/typing_screen.cpp)
target_link_libraries(dashtype PRIVATE ftxui::screen ftxui::dom ftxui::component)
