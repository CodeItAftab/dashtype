cmake_minimum_required(VERSION 3.16)
project(dashtype LANGUAGES CXX C)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

include(FetchContent)

FetchContent_Declare(
  ftxui
  GIT_REPOSITORY https://github.com/ArthurSonzogni/FTXUI.git
  GIT_TAG v5.0.0
)
FetchContent_MakeAvailable(ftxui)

FetchContent_Declare(
  sqlite3
  URL https://www.sqlite.org/2024/sqlite-amalgamation-3450100.zip
)
FetchContent_MakeAvailable(sqlite3)
add_library(sqlite3lib STATIC ${sqlite3_SOURCE_DIR}/sqlite3.c)
target_include_directories(sqlite3lib PUBLIC ${sqlite3_SOURCE_DIR})

find_package(Threads REQUIRED)

# Platform-specific networking backend: WinHTTP on Windows (no external
# dependency needed), libcurl on Linux (needs libcurl4-openssl-dev installed).
if(WIN32)
  set(PLATFORM_DOWNLOADER_SRC src/downloader_win.cpp)
else()
  find_package(CURL REQUIRED)
  set(PLATFORM_DOWNLOADER_SRC src/downloader_linux.cpp)
endif()

set(DASHTYPE_CORE_SOURCES
  src/cli_parser.cpp
  src/text_buffer.cpp
  src/typing_engine.cpp
  src/metrics.cpp
  src/typing_screen.cpp
  src/results_screen.cpp
  src/home_screen.cpp
  src/dataset_manager.cpp
  src/database.cpp
  src/config_store.cpp
  src/dataset_meta.cpp
  src/app_paths.cpp
  ${PLATFORM_DOWNLOADER_SRC}
)

add_executable(dashtype src/main.cpp ${DASHTYPE_CORE_SOURCES})
target_include_directories(dashtype PRIVATE src)

target_link_libraries(dashtype
  PRIVATE ftxui::screen ftxui::dom ftxui::component
          sqlite3lib Threads::Threads
)

if(WIN32)
  target_link_libraries(dashtype PRIVATE winhttp)
else()
  target_link_libraries(dashtype PRIVATE CURL::libcurl)
endif()

add_custom_command(TARGET dashtype POST_BUILD
  COMMAND ${CMAKE_COMMAND} -E copy_directory
      ${CMAKE_SOURCE_DIR}/data
      $<TARGET_FILE_DIR:dashtype>/data
)

install(TARGETS dashtype RUNTIME DESTINATION .)
install(DIRECTORY data/ DESTINATION data)

set(CPACK_PACKAGE_NAME "dashtype")
set(CPACK_PACKAGE_VERSION "0.1.0")
if(WIN32)
  set(CPACK_GENERATOR "ZIP")
else()
  set(CPACK_GENERATOR "TGZ")
endif()
include(CPack)

# --- Tests ---------------------------------------------------------------
enable_testing()

add_executable(metrics_test tests/metrics_test.cpp src/metrics.cpp src/typing_engine.cpp)
target_include_directories(metrics_test PRIVATE src)
add_test(NAME metrics_test COMMAND metrics_test)

add_executable(typing_engine_test tests/typing_engine_test.cpp src/typing_engine.cpp)
target_include_directories(typing_engine_test PRIVATE src)
add_test(NAME typing_engine_test COMMAND typing_engine_test)
